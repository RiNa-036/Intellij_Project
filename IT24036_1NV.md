# Intellij_Project

Prime JDBC + Servlet Assignment
Requirements
JDK 17 or later
IntelliJ IDEA Ultimate
MySQL Server + MySQL Workbench
Apache Tomcat 10.1.x
1. Create the database
Open MySQL Workbench and run database.sql.

2. Set your MySQL login
Open src/main/java/com/assignment/util/DBConnection.java and update:

USER
PASSWORD
The default database name is prime_assignment.

3. Open in IntelliJ
Extract this ZIP.
IntelliJ -> File -> Open -> select the PrimeJDBCServletAssignment folder.
Trust the project and wait for Maven dependencies to download.
Set Project SDK to JDK 17 or later.
4. Task 1: Insert nth prime using CMD
Run com.assignment.task1.PrimeInsertApp from IntelliJ, or build the project and run it from CMD.

From IntelliJ, provide the value of n in the Run configuration's Program arguments, for example: 100

From CMD after building:

mvn clean package
java -cp "target/classes;YOUR_MYSQL_CONNECTOR_JAR" com.assignment.task1.PrimeInsertApp 100
Replace YOUR_MYSQL_CONNECTOR_JAR with the path to the MySQL Connector/J jar if running directly from CMD. The easier option is to run the main class from IntelliJ; Maven manages the connector dependency.

The program calculates the nth prime and inserts n and the prime number into prime_numbers. The primary key id is generated automatically by MySQL.

5. Task 3 and Task 4: Run servlets in Tomcat
Add Tomcat 10.1 in IntelliJ: Run -> Edit Configurations -> + -> Tomcat Server -> Local.
In the Deployment tab, click + and select PrimeJDBCServletAssignment:war exploded.
Set the application context to /PrimeJDBCServletAssignment.
Apply and Run Tomcat.
Browser URLs:

HelloWorld form: http://localhost:8080/PrimeJDBCServletAssignment/hello
Browser primality checker: http://localhost:8080/PrimeJDBCServletAssignment/prime-check
Both pages take a number and show whether it is prime in a result field.

Notes
This project uses jakarta.servlet, so use Tomcat 10.1 (not Tomcat 9).
Task 1 inserts the nth prime into MySQL. Example: input 100 inserts n = 100 and prime_number = 541.
Task 3 and Task 4 are separate servlet URLs, but both use the same prime-checking utility.

HelloWorldServlet:-
package com.assignment.servlet;

import com.assignment.util.PrimeUtil;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import java.io.IOException;
import java.io.PrintWriter;

@WebServlet("/hello")
public class HelloWorldServlet extends HttpServlet {

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws IOException {
        showPage(request, response, null, "");
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws IOException {
        String numberText = request.getParameter("number");
        String result = "";

        try {
            long number = Long.parseLong(numberText);
            result = number + (PrimeUtil.isPrime(number) ? " is a Prime Number"
                    : " is Not a Prime Number");
        } catch (NumberFormatException e) {
            result = "Please enter a valid whole number.";
        }

        showPage(request, response, numberText, result);
    }

    private void showPage(HttpServletRequest request, HttpServletResponse response,
                          String number, String result) throws IOException {
        response.setContentType("text/html;charset=UTF-8");

        try (PrintWriter out = response.getWriter()) {
            out.println("<!DOCTYPE html><html><head><title>HelloWorld Servlet</title>");
            out.println("<style>body{font-family:Arial;margin:50px;background:#f4f6f8}"
                    + ".box{background:white;padding:25px;max-width:430px;border-radius:10px}"
                    + "input{padding:10px;width:95%;margin:8px 0;box-sizing:border-box}"
                    + "button{padding:10px 18px;background:#2457a7;color:white;border:0}"
                    + "</style></head><body><div class='box'>");
            out.println("<h2>Hello World Servlet</h2>");
            out.println("<form method='post'>");
            out.println("<label>Enter number:</label>");
            out.println("<input type='number' name='number' required value='"
                    + escape(number == null ? "" : number) + "'>");
            out.println("<label>Primality Check:</label>");
            out.println("<input type='text' readonly value='"
                    + escape(result) + "'>");
            out.println("<button type='submit'>Check</button></form>");
            out.println("</div></body></html>");
        }
    }

    private String escape(String value) {
        if (value == null) return "";
        return value.replace("&", "&amp;").replace("\"", "&quot;")
                .replace("<", "&lt;").replace(">", "&gt;");
    }
}

PrimeInsertApp:-
package com.assignment.task1;

import com.assignment.util.DBConnection;
import com.assignment.util.PrimeUtil;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.Statement;
import java.util.Scanner;

public class PrimeInsertApp {

    public static void main(String[] args) {
        try {
            int n;

            if (args.length > 0) {
                n = Integer.parseInt(args[0]);
            } else {
                Scanner scanner = new Scanner(System.in);
                System.out.print("Enter n: ");
                n = scanner.nextInt();
            }

            if (n < 1) {
                System.out.println("Please enter a positive integer.");
                return;
            }

            long prime = PrimeUtil.nthPrime(n);
            String sql = "INSERT INTO prime_numbers (n, prime_number) VALUES (?, ?)";

            try (Connection connection = DBConnection.getConnection();
                 PreparedStatement statement =
                         connection.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS)) {

                statement.setInt(1, n);
                statement.setLong(2, prime);
                statement.executeUpdate();

                try (ResultSet keys = statement.getGeneratedKeys()) {
                    if (keys.next()) {
                        System.out.println("Inserted successfully!");
                        System.out.println("Auto-generated ID: " + keys.getLong(1));
                    }
                }

                System.out.println("The " + n + "th prime number is: " + prime);
            }

        } catch (NumberFormatException e) {
            System.out.println("Invalid input. Please enter an integer.");
        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
            e.printStackTrace();
        }
    }
}

PrimeUtil:-
package com.assignment.util;

public class PrimeUtil {

    public static boolean isPrime(long number) {
        if (number < 2) {
            return false;
        }

        for (long i = 2; i <= number / i; i++) {
            if (number % i == 0) {
                return false;
            }
        }
        return true;
    }

    public static long nthPrime(int n) {
        if (n < 1) {
            throw new IllegalArgumentException("n must be at least 1.");
        }

        int count = 0;
        long number = 1;

        while (count < n) {
            number++;
            if (isPrime(number)) {
                count++;
            }
        }
        return number;
    }
}
